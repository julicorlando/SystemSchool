3. CONTADOR NO DASHBOARD DO ALUNO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   ┌─────────────────────┐
   │ Aluno acessa        │
   │ dashboard_aluno.php │
   └──────────┬──────────┘
              │
              ▼
   ┌──────────────────────────────────────────┐
   │ Query de contagem:                       │
   │                                          │
   │ SELECT COUNT(*) FROM atividades a        │
   │ WHERE a.turma_id = ?                     │
   │   AND (a.tipo='atividade' OR tipo IS NULL)│  ◄── Filtra tipo
   │   AND NOT EXISTS (                       │
   │       SELECT 1 FROM respostas r          │
   │       WHERE r.atividade_id=a.id          │  ◄── Verifica resposta
   │         AND r.aluno_id=?                 │
   │   )                                      │
   └──────────┬───────────────────────────────┘
              │
              ▼
   ┌──────────────────────┐
   │ pendentes = N        │
   └──────────┬───────────┘
              │
              ▼
   ┌──────────────────────────────────┐
   │ N > 0 ?                          │
   └──────────┬──────────┬────────────┘
              │          │
           SIM │          │ NÃO
              ▼          ▼
   ┌──────────────┐   ┌──────────────┐
   │ Mostra badge │   │ Não mostra   │
   │ vermelho: N  │   │ badge        │
   └──────────────┘   └──────────────┘
